SCALE-FREE NETWORK (Barabási-Albert model):
- Preferential attachment: new nodes connect to high-degree nodes
- Power-law degree distribution (realistic for social networks)

- High clustering coefficient + short path lengths
- Realistic for real-world social networks
- Implementation: generate_small_world_network(n: int, k: int, p: float) -> NetworkTopology

COMMUNITY-BASED NETWORK:
- Generate communities first, then connect them
- Within-community density > between-community density
- Implementation: generate_community_network(num_communities: int,

LINKEDIN-STYLE NETWORK:
- Professional connections based on industry, company, role
- Endorsements and recommendations as weighted edges
- Implementation: generate_professional_network(personas: List[TinyPerson]) -> NetworkTopology

class Neo4jNetworkStore:
    def __init__(self, uri: str, user: str, password: str):
        self.driver = GraphDatabase.driver(uri, auth=(user, password))
    
    def store_network(self, network: NetworkTopology) -> None:
        """Persist network to Neo4j"""
        with self.driver.session() as session:
            # Create persona nodes
            for persona in network.nodes.values():
                session.run(
                    "CREATE (p:Persona {id: $id, name: $name, ...}) ",
                    id=persona.id,
                    name=persona.name,
                    # ... other properties
                )
            
            # Create connection edges
            for connection in network.edges:
                session.run(
                    "MATCH (a:Persona {id: $source}), (b:Persona {id: $target}) "
                    "CREATE (a)-[r:CONNECTED {strength: $strength, ...}]->(b)",
                    source=connection.source_id,
                    target=connection.target_id,
                    strength=connection.strength
                )
    
    def query_neighbors(self, persona_id: str, depth: int = 1) -> List[Dict]:
        """Query network neighbors efficiently"""
        with self.driver.session() as session:
            result = session.run(
                f"MATCH (p:Persona {{id: $pid}})-[*1..{depth}]-(neighbor) "
                "RETURN neighbor",
                pid=persona_id
            )
            return [record["neighbor"] for record in result]
    
    def find_influence_path(self, source_id: str, target_id: str) -> List[str]:
        """Find shortest influence path between two personas"""
        with self.driver.session() as session:
            result = session.run(
                "MATCH path = shortestPath((a:Persona {id: $source})-[*]-(b:Persona {id: $target})) "
                "RETURN path",
                source=source_id,
                target=target_id
            )
            return self._extract_path(result.single()["path"])
    
    def get_community_members(self, community_id: str) -> List[str]:

class EngagementPredictor:
    """Predicts whether persona will engage with content"""
    
    def __init__(self):
        self.model = None  # Trained model
        self.feature_extractor = FeatureExtractor()
        self.scaler = StandardScaler()
    
    def train(self, training_data: List[TrainingExample]) -> None:
        """Train the prediction model"""
        # Extract features
        X = []
        y = []
        for example in training_data:
            features = self._extract_all_features(
                example.persona,
                example.content,
                example.network
            )
            X.append(features)
            y.append(1 if example.engaged else 0)
        
        X = np.array(X)
        y = np.array(y)
        
        # Scale features
        X_scaled = self.scaler.fit_transform(X)
        
        # Train model (try multiple algorithms)
        models = {
            "logistic": LogisticRegression(class_weight="balanced"),
            "random_forest": RandomForestClassifier(n_estimators=100, class_weight="balanced"),
            "xgboost": XGBClassifier(scale_pos_weight=self._calculate_pos_weight(y)),
            "lightgbm": LGBMClassifier(is_unbalance=True),
        }
        
        # Cross-validation to select best model
        best_score = 0
        best_model = None
        for name, model in models.items():
            scores = cross_val_score(model, X_scaled, y, cv=5, scoring="f1")
            avg_score = scores.mean()
            if avg_score > best_score:
                best_score = avg_score
                best_model = model
        
        # Train final model on all data
        self.model = best_model
        self.model.fit(X_scaled, y)
    
    def predict(self, persona: TinyPerson, content: Content, network: NetworkTopology) -> float:
        """Predict engagement probability"""
        features = self._extract_all_features(persona, content, network)
        features_scaled = self.scaler.transform([features])
        probability = self.model.predict_proba(features_scaled)[0][1]
        return probability
    
    def _extract_all_features(self, persona, content, network) -> np.ndarray:
        """Combine all feature extractors"""
        content_features = ContentFeatureExtractor().extract(content)
        persona_features = PersonaFeatureExtractor().extract(persona)
        interaction_features = InteractionFeatureExtractor().extract(persona, content, network)
        
        all_features = {**content_features, **persona_features, **interaction_features}
        return np.array(list(all_features.values()))

class EnsemblePredictor:
    """Combines multiple predictors for robust predictions"""
    
    def __init__(self):
        self.engagement_predictor = EngagementPredictor()
        self.type_predictor = EngagementTypePredictor()
        self.virality_predictor = ViralityPredictor()
        self.llm_predictor = LLMPredictor()  # Use LLM as another signal
    
    def predict(self, persona: TinyPerson, content: Content, 
                network: NetworkTopology) -> PredictionResult:
        """Ensemble prediction"""
        # Get predictions from all models
        ml_prob = self.engagement_predictor.predict(persona, content, network)
        type_probs = self.type_predictor.predict(persona, content)
        llm_prediction = self.llm_predictor.predict(persona, content)
        
        # Weighted combination
        ensemble_prob = (
            0.5 * ml_prob +
            0.3 * type_probs.get("any_engagement", 0.0) +
            0.2 * llm_prediction.probability
        )
        
        return PredictionResult(
            engagement_probability=ensemble_prob,
            engagement_type_probs=type_probs,
            predicted_reaction=self._select_reaction(type_probs),
            confidence=self._calculate_confidence([ml_prob, type_probs, llm_prediction])
        )

class LLMPredictor:
    """Use LLM reasoning for engagement prediction"""
    
    def __init__(self, model: str = "gpt-4"):
        self.model = model
        self.client = OpenAI()
    
    def predict(self, persona: TinyPerson, content: Content) -> LLMPrediction:
        """Use LLM to predict engagement"""
        
        prompt = self._construct_prediction_prompt(persona, content)
        
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": PREDICTION_SYSTEM_PROMPT},
                {"role": "user", "content": prompt}
            ],
            temperature=0.3,
            response_format={"type": "json_object"}
        )
        
        prediction = json.loads(response.choices[0].message.content)
        
        return LLMPrediction(
            will_engage=prediction["will_engage"],
            probability=prediction["probability"],
            reasoning=prediction["reasoning"],
            predicted_reaction_type=prediction["reaction_type"],

class VariantOptimizer:
    """Optimize and rank content variants"""
    
    def __init__(self, predictor: EngagementPredictor):
        self.predictor = predictor
    
    def rank_variants_for_audience(self, variants: List[ContentVariant],
                                   target_personas: List[TinyPerson],
                                   network: NetworkTopology,
                                   objective: str = "maximize_engagement") -> List[RankedVariant]:
        """Rank variants by predicted performance"""
        
        ranked = []
        
        for variant in variants:
            # Predict engagement for each persona
            engagement_probs = []
            for persona in target_personas:
                prob = self.predictor.predict(
                    persona,
                    Content(text=variant.text),
                    network
                )
                engagement_probs.append(prob)
            
            # Calculate aggregate metrics based on objective
            if objective == "maximize_engagement":
                score = np.mean(engagement_probs)
            elif objective == "maximize_reach":
                # Weight by persona influence
                weighted_probs = [
                    prob * persona.influence_metrics.reach 
                    for prob, persona in zip(engagement_probs, target_personas)
                ]
                score = np.mean(weighted_probs)
            elif objective == "maximize_conversions":
                # Focus on high-intent personas
                high_intent_personas = [p for p in target_personas if p.intent_score > 0.7]
                relevant_probs = [engagement_probs[i] for i, p in enumerate(target_personas) 
                                if p in high_intent_personas]
                score = np.mean(relevant_probs) if relevant_probs else 0.0
            
            ranked.append(RankedVariant(
                variant=variant,
                score=score,
                engagement_probs=engagement_probs,
                predicted_reach=self._estimate_reach(variant, target_personas, network),
                predicted_engagement_count=int(sum(engagement_probs))
            ))
        
        # Sort by score
        ranked.sort(key=lambda x: x.score, reverse=True)
        
        return ranked
    
    def _estimate_reach(self, variant: ContentVariant, 
                       personas: List[TinyPerson],
                       network: NetworkTopology) -> int:
        """Estimate total reach through cascade simulation"""
        
        # Simulate cascade for top engaging personas
        engaging_personas = [p for p, prob in zip(personas, engagement_probs) 
                           if prob > 0.5]
        
        if not engaging_personas:
            return 0
        
        propagator = InfluencePropagator(network)
        result = propagator.propagate(
            [p.id for p in engaging_personas[:10]],  # Top 10 seed nodes
            Content(text=variant.text)
        )
        
        return result.total_reach
    
    def generate_variant_report(self, ranked_variants: List[RankedVariant],
                               top_k: int = 10) -> VariantReport:
        """Generate report comparing top variants"""
        
        top_variants = ranked_variants[:top_k]
        
        return VariantReport(
            top_variants=top_variants,
            best_variant=top_variants[0],
            score_distribution=[v.score for v in ranked_variants],
            strategy_performance=self._analyze_strategy_performance(ranked_variants),
            recommendations=self._generate_recommendations(top_variants)
        )
    
    def _analyze_strategy_performance(self, variants: List[RankedVariant]) -> Dict:
        """Analyze which generation strategies work best"""
        
        strategy_scores = defaultdict(list)
        for ranked_variant in variants:
            strategy = ranked_variant.variant.strategy
            strategy_scores[strategy].append(ranked_variant.score)
        
        return {
            strategy: {
                "avg_score": np.mean(scores),
                "top_10_percentage": sum(1 for v in variants[:10] 
                                       if v.variant.strategy == strategy) / 10
            }

        }

class ABTestSimulator:
    """Simulate A/B tests to compare content variants"""
    
    def run_test(self, variant_a: ContentVariant, variant_b: ContentVariant,
                 audience: List[TinyPerson], network: NetworkTopology,
                 duration_days: int = 7) -> ABTestResult:
        """Simulate A/B test between two variants"""
        
        # Split audience randomly
        split_point = len(audience) // 2
        audience_a = audience[:split_point]
        audience_b = audience[split_point:]
        
        # Simulate engagement for each group
        results_a = self._simulate_engagement(variant_a, audience_a, network, duration_days)
        results_b = self._simulate_engagement(variant_b, audience_b, network, duration_days)
        
        # Statistical comparison
        significance = self._calculate_significance(results_a, results_b)
        
        return ABTestResult(
            variant_a=variant_a,
            variant_b=variant_b,
            engagement_rate_a=results_a.engagement_rate,
            engagement_rate_b=results_b.engagement_rate,
            winner="A" if results_a.engagement_rate > results_b.engagement_rate else "B",
            lift_percentage=self._calculate_lift(results_a, results_b),
            statistical_significance=significance,
            confidence_level=self._calculate_confidence(significance)
        )
    
    def _simulate_engagement(self, variant: ContentVariant, 
                            audience: List[TinyPerson],
                            network: NetworkTopology,
                            duration_days: int) -> EngagementResults:
        """Simulate engagement over time"""
        
        engaged_count = 0
        engagements = []
        
        for persona in audience:
            # Predict engagement
            prob = self.predictor.predict(persona, Content(text=variant.text), network)
            engaged = random.random() < prob
            
            if engaged:
                engaged_count += 1
                engagement_time = random.randint(0, duration_days * 24 * 60)  # Random time in minutes
                engagements.append(Engagement(
                    persona_id=persona.id,
                    timestamp=engagement_time,
                    type=self._sample_engagement_type(persona)
                ))
        
        return EngagementResults(
            total_views=len(audience),
            total_engagements=engaged_count,
            engagement_rate=engaged_count / len(audience),
            engagements=engagements,
            cascade_reach=self._simulate_cascade(variant, engagements, network)
        )
    
    def _calculate_significance(self, results_a: EngagementResults, 
                                results_b: EngagementResults) -> float:
        """Calculate statistical significance using chi-square test"""
        
        from scipy.stats import chi2_contingency
        
        contingency_table = [
            [results_a.total_engagements, results_a.total_views - results_a.total_engagements],
            [results_b.total_engagements, results_b.total_views - results_b.total_engagements]
        ]
        
        chi2, p_value, dof, expected = chi2_contingency(contingency_table)
        
        return p_value

class LinkedInAuth:
    """Handle LinkedIn OAuth authentication"""
    
    def __init__(self, client_id: str, client_secret: str, redirect_uri: str):
        self.client_id = client_id
        self.client_secret = client_secret
        self.redirect_uri = redirect_uri
        self.base_url = "https://www.linkedin.com/oauth/v2"
        self.api_base = "https://api.linkedin.com/v2"
    
    def get_authorization_url(self, state: str) -> str:
        """Generate LinkedIn authorization URL"""
        
        scopes = [
            "r_liteprofile",
            "r_emailaddress",
            "w_member_social",
            "r_organization_social",
            "rw_organization_admin"
        ]
        
        params = {
            "response_type": "code",
            "client_id": self.client_id,
            "redirect_uri": self.redirect_uri,
            "state": state,
            "scope": " ".join(scopes)
        }
        
        return f"{self.base_url}/authorization?" + urlencode(params)
    
    def exchange_code_for_token(self, code: str) -> AccessToken:
        """Exchange authorization code for access token"""
        
        data = {
            "grant_type": "authorization_code",
            "code": code,
            "redirect_uri": self.redirect_uri,
            "client_id": self.client_id,
            "client_secret": self.client_secret
        }
        
        response = requests.post(f"{self.base_url}/accessToken", data=data)
        response.raise_for_status()
        
        token_data = response.json()
        
        return AccessToken(
            access_token=token_data["access_token"],
            expires_in=token_data["expires_in"],
            refresh_token=token_data.get("refresh_token"),
            created_at=datetime.now()
        )
    
    def refresh_access_token(self, refresh_token: str) -> AccessToken:
        """Refresh expired access token"""
        
        data = {

            "refresh_token": refresh_token,
            "client_id": self.client_id,

        }

class LinkedInPostSimulator:
    """Simulate LinkedIn post performance before publishing"""
    
    def __init__(self, predictor: EngagementPredictor):
        self.predictor = predictor
    
    def simulate_post(self, post_content: str, 
                     audience_personas: List[TinyPerson],
                     network: NetworkTopology) -> PostSimulationResult:
        """Simulate post performance on LinkedIn"""
        
        # Create Content object
        content = Content(
            text=post_content,
            platform="LinkedIn",
            content_type="post",
            timestamp=datetime.now()
        )
        
        # Predict engagement for each persona
        engagement_predictions = []
        for persona in audience_personas:
            prediction = self.predictor.predict(persona, content, network)
            engagement_predictions.append(EngagementPrediction(
                persona_id=persona.id,
                will_engage=prediction.engagement_probability > 0.5,
                engagement_probability=prediction.engagement_probability,
                predicted_reaction=prediction.predicted_reaction,
                predicted_comment=prediction.predicted_comment
            ))
        
        # Aggregate metrics
        total_reach = len(audience_personas)
        expected_likes = sum(1 for p in engagement_predictions if p.predicted_reaction == "like")
        expected_comments = sum(1 for p in engagement_predictions if p.predicted_reaction == "comment")
        expected_shares = sum(1 for p in engagement_predictions if p.predicted_reaction == "share")
        
        # Simulate viral spread
        viral_reach = self._simulate_viral_spread(
            engagement_predictions, 
            network, 
            content
        )
        
        # Calculate engagement rate
        total_engagements = expected_likes + expected_comments + expected_shares
        engagement_rate = total_engagements / total_reach if total_reach > 0 else 0.0
        
        return PostSimulationResult(
            post_content=post_content,
            direct_reach=total_reach,
            viral_reach=viral_reach,
            total_reach=total_reach + viral_reach,
            expected_likes=expected_likes,
            expected_comments=expected_comments,
            expected_shares=expected_shares,
            engagement_rate=engagement_rate,
            engagement_predictions=engagement_predictions,
            confidence_score=self._calculate_confidence(engagement_predictions)
        )
    
    def _simulate_viral_spread(self, initial_engagements: List[EngagementPrediction],
                              network: NetworkTopology,
                              content: Content) -> int:
        """Simulate secondary engagement through shares"""
        
        # Find personas who will share
        sharers = [p for p in initial_engagements if p.predicted_reaction == "share"]
        
        if not sharers:
            return 0
        
        # Simulate cascade from sharers
        propagator = InfluencePropagator(network)
        result = propagator.propagate(
            [p.persona_id for p in sharers],
            content
        )
        
        # Exclude original audience
        viral_reach = result.total_reach - len(initial_engagements)
        
        return max(0, viral_reach)
    
    def compare_post_variants(self, variants: List[str],
                            audience: List[TinyPerson],
                            network: NetworkTopology) -> ComparisonResult:
        """Compare multiple post variants"""
        
        results = []
        for variant_text in variants:
            result = self.simulate_post(variant_text, audience, network)
            results.append(result)
        
        # Sort by engagement rate
        results.sort(key=lambda x: x.engagement_rate, reverse=True)
        
        return ComparisonResult(
            best_variant=results[0],
            all_results=results,
            winner_improvement=self._calculate_improvement(results[0], results[-1])
        )

class SocialTinyWorld(TinyWorld):
    """Extended TinyWorld with social network capabilities"""
    
    def __init__(self, name: str, network: NetworkTopology = None):
        super().__init__(name)
        self.network = network or NetworkTopology()
        self.content_items = []
        self.simulation_history = []
        self.time_step = 0
    
    def add_content(self, content: Content) -> None:
        """Add content to the world for personas to interact with"""
        self.content_items.append(content)
        self.broadcast_to_agents(f"New content available: {content.text[:100]}...")
    
    def simulate_content_spread(self, content: Content, 
                               initial_viewers: List[str],
                               max_steps: int = 10) -> SimulationResult:
        """Simulate how content spreads through the network"""
        
        result = SimulationResult(
            content=content,
            start_time=datetime.now()
        )
        
        # Initialize: Initial viewers see content
        viewed = set(initial_viewers)
        engaged = set()
        
        for step in range(max_steps):
            self.time_step = step
            new_viewers = set()
            
            # Each viewer decides whether to engage
            for viewer_id in viewed - engaged:
                persona = self.network.nodes[viewer_id]
                
                # Simulate persona viewing and deciding
                decision = self._simulate_engagement_decision(persona, content, step)
                
                if decision.engaged:
                    engaged.add(viewer_id)
                    result.add_engagement(viewer_id, decision.engagement_type, step)
                    
                    # If shared, content spreads to their connections
                    if decision.engagement_type == "share":
                        neighbors = self.network.get_neighbors(viewer_id)
                        new_viewers.update([n.id for n in neighbors])
            
            # Add new viewers
            viewed.update(new_viewers)
            
            # Track metrics for this step
            result.add_step_metrics(step, len(viewed), len(engaged))
            
            # Stop if no new activity
            if not new_viewers and not any(v not in engaged for v in viewed):
                break
        
        result.finalize(datetime.now())
        self.simulation_history.append(result)
        
        return result
    
    def _simulate_engagement_decision(self, persona: TinyPerson, 
                                     content: Content,
                                     time_step: int) -> EngagementDecision:
        """Simulate a persona's decision to engage with content"""
        
        # Gather context
        network_influence = self._calculate_network_influence(persona, content)
        content_relevance = persona.get_content_affinity(content)
        
        # Generate engagement probability
        base_prob = persona.calculate_engagement_probability(content)
        
        # Adjust for network influence
        adjusted_prob = base_prob * (1 + 0.5 * network_influence)
        
        # Adjust for time decay (content gets stale)
        time_decay = 0.9 ** time_step
        final_prob = adjusted_prob * time_decay
        
        # Make decision
        engaged = random.random() < final_prob
        
        if engaged:
            # Determine engagement type
            engagement_type = self._sample_engagement_type(persona, content)
            
            # Generate comment if applicable
            comment = None
            if engagement_type == "comment":
                comment = self._generate_comment(persona, content)
            
            return EngagementDecision(
                engaged=True,
                engagement_type=engagement_type,
                comment=comment,
                probability=final_prob
            )
        
        return EngagementDecision(engaged=False, probability=final_prob)
    
    def _calculate_network_influence(self, persona: TinyPerson, 
                                    content: Content) -> float:
        """Calculate social influence on engagement decision"""
        
        # Check if persona's connections have engaged
        connections = self.network.get_neighbors(persona.id)
        engaged_connections = [
            c for c in connections 
            if self._has_engaged_with_content(c.id, content)
        ]
        
        if not connections:
            return 0.0
        
        # Weight by connection strength
        influence_sum = sum(
            self.network.get_connection_strength(persona.id, c.id)
            for c in engaged_connections
        )
        
        # Normalize
        influence = influence_sum / len(connections)
        
        return influence
    
    def run_parallel_simulations(self, scenarios: List[Scenario]) -> List[SimulationResult]:
        """Run multiple simulation scenarios in parallel"""
        
        from concurrent.futures import ThreadPoolExecutor
        
        with ThreadPoolExecutor(max_workers=10) as executor:
            futures = [
                executor.submit(self._run_scenario, scenario)
                for scenario in scenarios
            ]
            
            results = [future.result() for future in futures]
        
        return results
    
    def _run_scenario(self, scenario: Scenario) -> SimulationResult:
        """Run a single simulation scenario"""
        
        # Reset state for this scenario
        world_copy = self._create_world_copy()
        
        # Run simulation
        result = world_copy.simulate_content_spread(
            scenario.content,
            scenario.initial_viewers,
            scenario.max_steps
        )
        
        return result

class SimulationManager:
    """Manages simulation lifecycle and execution"""
    
    def __init__(self):
        self.worlds = {}
        self.predictor = EngagementPredictor()
        self.variant_generator = ContentVariantGenerator()
    
    def create_simulation(self, config: SimulationConfig) -> Simulation:
        """Create a new simulation from configuration"""
        
        # Generate or load personas
        if config.use_linkedin_audience:
            personas = self._create_personas_from_linkedin(config.linkedin_token)
        else:
            personas = self._generate_synthetic_personas(config.persona_count)
        
        # Generate network
        network = self._create_network(personas, config.network_type)
        
        # Create world
        world = SocialTinyWorld(config.name, network)
        for persona in personas:
            world.add_agent(persona)
        
        # Create simulation object
        simulation = Simulation(
            id=generate_id(),
            config=config,
            world=world,
            personas=personas,
            network=network,
            status="ready"
        )
        
        self.worlds[simulation.id] = simulation
        
        return simulation
    
    def run_simulation(self, simulation_id: str, content: Content,
                      mode: str = "full") -> SimulationResult:
        """Run a simulation"""
        
        simulation = self.worlds[simulation_id]
        simulation.status = "running"
        
        try:
            if mode == "full":
                # Run full cascade simulation
                result = simulation.world.simulate_content_spread(
                    content,
                    initial_viewers=self._select_initial_viewers(simulation),
                    max_steps=10
                )
            elif mode == "fast":
                # Use prediction only (no cascade)
                result = self._fast_prediction_simulation(simulation, content)
            elif mode == "variants":
                # Generate and test multiple variants
                result = self._variant_simulation(simulation, content)
            
            simulation.status = "completed"
            simulation.last_result = result
            
            return result
            
        except Exception as e:
            simulation.status = "failed"
            simulation.error = str(e)
            raise
    
    def _fast_prediction_simulation(self, simulation: Simulation, 
                                   content: Content) -> SimulationResult:
        """Fast simulation using predictions only"""
        
        result = SimulationResult(content=content, start_time=datetime.now())
        
        for persona in simulation.personas:
            prob = self.predictor.predict(persona, content, simulation.network)
            
            if prob > 0.5:
                result.add_engagement(
                    persona.id,
                    self._sample_engagement_type(persona, content),
                    time_step=0
                )
        
        result.finalize(datetime.now())
        return result
    
    def _variant_simulation(self, simulation: Simulation,
                          content: Content) -> VariantSimulationResult:
        """Generate and test multiple content variants"""
        
        # Generate variants
        variants = self.variant_generator.generate_variants(
            content.text,
            num_variants=100,
            target_personas=simulation.personas[:20]  # Use sample for targeting
        )
        
        # Test each variant
        results = []
        for variant in variants:
            variant_content = Content(text=variant.text, **content.metadata)
            result = self._fast_prediction_simulation(simulation, variant_content)
            results.append((variant, result))
        
        # Rank by performance
        results.sort(key=lambda x: x[1].engagement_rate, reverse=True)
        
        return VariantSimulationResult(
            original_content=content,
            variants=variants,
            results=results,
            best_variant=results[0]
        )

class SimulationCache:
    """Cache simulation results for faster repeated simulations"""
    
    def __init__(self, redis_client: redis.Redis):
        self.redis = redis_client
        self.ttl = 3600 * 24 * 7  # 1 week
    
    def get_cached_prediction(self, persona_id: str, content_hash: str) -> Optional[float]:
        """Get cached engagement prediction"""
        
        key = f"prediction:{persona_id}:{content_hash}"
        cached = self.redis.get(key)
        
        if cached:
            return float(cached)
        
        return None
    
    def cache_prediction(self, persona_id: str, content_hash: str, 
                        probability: float) -> None:
        """Cache engagement prediction"""
        
        key = f"prediction:{persona_id}:{content_hash}"
        self.redis.setex(key, self.ttl, probability)
    
    def get_cached_simulation(self, config_hash: str) -> Optional[SimulationResult]:
        """Get cached simulation result"""
        
        key = f"simulation:{config_hash}"
        cached = self.redis.get(key)
        
        if cached:
            return pickle.loads(cached)
        
        return None
    
    def cache_simulation(self, config_hash: str, result: SimulationResult) -> None:
        """Cache simulation result"""
        
        key = f"simulation:{config_hash}"
        self.redis.setex(key, self.ttl, pickle.dumps(result))
    
    @staticmethod
    def hash_content(content: Content) -> str:
        """Generate hash for content"""
        
        content_str = f"{content.text}:{content.platform}:{content.content_type}"
        return hashlib.sha256(content_str.encode()).hexdigest()
    
    @staticmethod
    def hash_config(config: SimulationConfig) -> str:
        """Generate hash for simulation configuration"""
        
        config_str = json.dumps(config.to_dict(), sort_keys=True)
        return hashlib.sha256(config_str.encode()).hexdigest()
